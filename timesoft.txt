SELECT
    T.TABLE_NAME AS [Nama Tabel],
    C.COLUMN_NAME AS [Nama Kolom],
    C.DATA_TYPE AS [Tipe Data],
    C.CHARACTER_MAXIMUM_LENGTH AS [Max Length],
    C.IS_NULLABLE AS [Nullable],
    FK.CONSTRAINT_NAME AS [Nama Constraint FK],
    TC_REF.TABLE_NAME AS [Tabel Referensi (Target)], 
    KCU_REF.COLUMN_NAME AS [Kolom Referensi (Target)]

FROM
    INFORMATION_SCHEMA.TABLES T
INNER JOIN
    INFORMATION_SCHEMA.COLUMNS C 
    ON T.TABLE_NAME = C.TABLE_NAME AND T.TABLE_SCHEMA = C.TABLE_SCHEMA
LEFT JOIN
    INFORMATION_SCHEMA.REFERENTIAL_CONSTRAINTS FK 
    ON C.TABLE_SCHEMA = FK.CONSTRAINT_SCHEMA
LEFT JOIN
    INFORMATION_SCHEMA.KEY_COLUMN_USAGE KCU 
    ON C.TABLE_NAME = KCU.TABLE_NAME 
    AND C.COLUMN_NAME = KCU.COLUMN_NAME 
    AND KCU.CONSTRAINT_NAME = FK.CONSTRAINT_NAME
    AND KCU.TABLE_SCHEMA = T.TABLE_SCHEMA
LEFT JOIN
    INFORMATION_SCHEMA.TABLE_CONSTRAINTS TC_REF
    ON TC_REF.CONSTRAINT_NAME = FK.UNIQUE_CONSTRAINT_NAME
    AND TC_REF.CONSTRAINT_SCHEMA = FK.UNIQUE_CONSTRAINT_SCHEMA
    AND TC_REF.CONSTRAINT_TYPE IN ('PRIMARY KEY', 'UNIQUE')
LEFT JOIN
    INFORMATION_SCHEMA.KEY_COLUMN_USAGE KCU_REF
    ON KCU_REF.CONSTRAINT_NAME = FK.UNIQUE_CONSTRAINT_NAME
    AND KCU_REF.TABLE_SCHEMA = FK.UNIQUE_CONSTRAINT_SCHEMA
    AND KCU_REF.ORDINAL_POSITION = KCU.ORDINAL_POSITION 

WHERE
    T.TABLE_SCHEMA = 'dbo' 
    AND T.TABLE_TYPE = 'BASE TABLE' 
ORDER BY
    T.TABLE_NAME, 
    C.ORDINAL_POSITION;
